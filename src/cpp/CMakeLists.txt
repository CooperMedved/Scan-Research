cmake_minimum_required(VERSION 3.16)
project(scan_partition_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Boost REQUIRED)
add_executable(scan_features feature_engine.cpp)
target_include_directories(scan_features PRIVATE ${Boost_INCLUDE_DIRS})
add_executable(scan_partition partition_main.cpp partition.cpp refinement.cpp)
foreach(target scan_features scan_partition)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -O3 -Wall -Wextra -Wpedantic)
  endif()
endforeach()
